cmake_minimum_required(VERSION 3.16)
project(cooperative_avoidance VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
include(GNUInstallDirs)
option(BUILD_SIMULATOR "Build JSON scenario runner and simulator" ON)
add_library(planning_core
  planning_core/src/geometry.cpp planning_core/src/reference_path.cpp
  planning_core/src/motion_model.cpp planning_core/src/trajectory.cpp
  planning_core/src/prediction.cpp planning_core/src/collision_checker.cpp
  planning_core/src/path_deformation.cpp planning_core/src/cooperative_planner.cpp)
add_library(cooperative_avoidance::planning_core ALIAS planning_core)
target_compile_features(planning_core PUBLIC cxx_std_17)
set_target_properties(planning_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(planning_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/planning_core/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_options(planning_core PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_SIMULATOR)
find_package(nlohmann_json 3.2 REQUIRED)
add_library(simulator simulator/src/scenario.cpp simulator/src/controller.cpp simulator/src/engine.cpp simulator/src/observations.cpp)
target_include_directories(simulator PUBLIC simulator/include)
target_link_libraries(simulator PUBLIC planning_core nlohmann_json::nlohmann_json)
target_compile_options(simulator PRIVATE -Wall -Wextra -Wpedantic)
add_executable(avoidance_sim simulator/main.cpp)
target_link_libraries(avoidance_sim PRIVATE simulator)
set_target_properties(avoidance_sim PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
install(TARGETS avoidance_sim RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(BUILD_TESTING)
  find_package(Boost REQUIRED)
  add_executable(core_tests tests/unit/core_tests.cpp)
  target_include_directories(core_tests PRIVATE ${Boost_INCLUDE_DIRS})
  target_link_libraries(core_tests PRIVATE planning_core)
  add_test(NAME core COMMAND core_tests)
  if(BUILD_SIMULATOR)
  add_executable(integration_tests tests/integration/scenario_tests.cpp)
  target_include_directories(integration_tests PRIVATE ${Boost_INCLUDE_DIRS})
  target_link_libraries(integration_tests PRIVATE simulator)
  target_compile_definitions(integration_tests PRIVATE SCENARIO_DIR="${CMAKE_CURRENT_SOURCE_DIR}/scenarios")
  add_test(NAME integration COMMAND integration_tests)
  find_package(Python3 3.10 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME editor COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/editor_tests.py
      --binary $<TARGET_FILE:avoidance_sim>)
    set_tests_properties(editor PROPERTIES ENVIRONMENT
      "PYTHONNOUSERSITE=1;MPLBACKEND=Agg;MPLCONFIGDIR=/tmp/path-planner-mpl")
  endif()
  endif()
endif()
install(TARGETS planning_core EXPORT planning_coreTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY planning_core/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT planning_coreTargets NAMESPACE cooperative_avoidance:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/planning_core)
install(FILES config/planning_coreConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/planning_core)
